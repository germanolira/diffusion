  step: number;
  label: string;

  label: string;
}

// Pure display, memoized to skip repeats.

  return (
    <p aria-live="polite">
      {label}: <strong>{value}</strong>
    </p>
  );
});

export function Counter({ initial, step, label }: CounterProps) {
  const [count, setCount] = useState<number>(initial);

  // Stable handlers keep memoized children cheap.
  const increment = useCallback(() => {
    setCount((c) => c + step);
  }, [step]);

  const decrement = useCallback(() => {
    setCount((c) => c - step);
  }, [step]);

  const reset = useCallback(() => {
    setCount(initial);
  }, [initial]);

  return (
    <section>
      <h2>Counter</h2>
      <CounterDisplay value={count} label={label} />
      <div>
        <button type="button" onClick={decrement}>
          Minus
        </button>
        <button type="button" onClick={increment}>
          Plus
        </button>
        <button type="button" onClick={reset}>
          Reset
        </button>
      </div>
    </section>
  );
}

import { useCallback, useEffect, useState } from 'react';

interface User {
  id: number;
  name: string;
}

interface DataFetcherProps {
  endpoint: string;
  pollMs: number;
}

type Status = 'idle' | 'loading' | 'error' | 'done';

async function fetchUsers(endpoint: string, signal: AbortSignal): Promise<User[]> {
  const res = await fetch(endpoint, { signal });
  if (!res.ok) throw new Error('Fetch failed');
  return (await res.json()) as User[];
}

export function DataFetcher({ endpoint, pollMs }: DataFetcherProps) {
  const [users, setUsers] = useState<User[]>([]);
  const [status, setStatus] = useState<Status>('idle');
  const [error, setError] = useState<string>('');

  const load = useCallback(async (endpoint: string, signal: AbortSignal) => {
    setStatus('loading');
    setError('');
    try {
      const data = await fetchUsers(endpoint, signal);
      setUsers(data);
      setStatus('done');
    } catch (err) {
      if (err instanceof DOMException && err.name === 'AbortError') return;
      setError(err instanceof Error ? err.message : 'Unknown error');
      setStatus('error');
    }
  }, []);

  // Fetch on mount plus optional polling.
  useEffect(() => {
    const controller = new AbortController();
    void load(endpoint, controller.signal);
    if (pollMs <= 0) return () => controller.abort();
    const timer = window.setInterval(() => {
      const c = new AbortController();
      void load(endpoint, c.signal);
    }, pollMs);
    return () => {
      controller.abort();
      window.clearInterval(timer);
    };
  }, [endpoint, pollMs, load]);

  if (status === 'loading' && users.length === 0) return <p>Loading users...</p>;
  if (status === 'error') return <p role="alert">Error: {error}</p>;

  return (
    <section>
      <h2>Users</h2>
      <ul>
        {users.map((u) => (
          <li key={u.id}>{u.name}</li>
        ))}
      </ul>
    </section>
  );
}

  hasError: boolean;
  message: string;
}

// Only class allowed: isolates render crashes.
export class ErrorBoundary extends Component<BoundaryProps, BoundaryState> {
  state: BoundaryState = { hasError: false, message: '' };

  static getDerivedStateFromError(error: Error): BoundaryState {
    return { hasError: true, message: error.message };
  }

  componentDidCatch(error: Error, info: ErrorInfo): void {
    console.error(error, info.componentStack);
  }

  reset = (): void => {
    this.setState({ hasError: false, message: '' });
  };

  render(): ReactNode {
    if (this.state.hasError) {
      return (
        <div role="alert">
          <h2>{this.props.fallbackTitle}</h2>
          <p>{this.state.message}</p>
          <button type="button" onClick={this.reset}>
            Try again
          </button>
        </div>
      );
    }
    return this.props.children;
  }
}

import { useEffect, useRef, useState } from 'react';
import type { ReactNode } from 'react';

interface ModalProps {
  title: string;
  triggerLabel: string;
  children: ReactNode;
}

export function Modal({ title, triggerLabel, children }: ModalProps) {
  const [open, setOpen] = useState<boolean>(false);
  const closeRef = useRef<HTMLButtonElement>(null);

  // Focus close button when modal opens.
  useEffect(() => {
    if (open) closeRef.current?.focus();
  }, [open ]);

  // Close on Escape for keyboard users.
  useEffect(() => {
    if (!open) return;
    function onKey(e: KeyboardEvent): void {
      if (e.key === 'Escape') setOpen(false);
    }
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [open ]);

  return (
    <div>
      <button type="button" onClick={() => setOpen(true)}>
        {triggerLabel}
      </button>
      {open ? (
        <div role="dialog" aria-modal="true" aria-label={title}>
          <div role="presentation" onClick={() => setOpen(false)} />
          <div>
            <h2>{title}</h2>
            <div>{children}</div>
            <button ref={closeRef} type="button" onClick={() => setOpen(false)}>
              Close
            </button>
          </div>
        </div>
      ) : null}
    </div>
  );
}

  initialPost: Post;
  label: string;
}

export function OptimisticLike({ initialPost, label }: OptimisticLikeProps) {
  const [post, setPost] = useState<Post>(initialPost);

  // Optimistic copy updates instantly, then syncs.
  const [optimistic, addOptimistic] = useOptimistic<Post, boolean>(
    post,
    (current, nextLiked) => ({
      ...current,
      liked: nextLiked,
      likes: current.likes + (nextLiked ? 1 : -1),
    })
  );

  async function toggle(): Promise<void> {
    const next = !optimistic.liked;
    addOptimistic(next);
    // Fake latency for the server write.
    await new Promise((r) => setTimeout(r, 400));
    setPost((p) => ({
      ...p,
      liked: next,
      likes: p.likes + (next ? 1 : -1),
    }));
  }

  return (
    <article>
      <h2>{optimistic.title}</h2>
      <p>
        {label}: {optimistic.likes}
      </p>
      <button type="button" onClick={toggle} aria-pressed={optimistic.liked}>
        {optimistic.liked ? 'Unlike' : 'Like'}
      </button>
    </article>
  );
}

export function TodoList({ title, initialTodos }: TodoListProps) {
  const [todos, setTodos] = useState<Todo[]>(initialTodos);
  const [draft, setDraft] = useState<string>('');

  // Derived state, no extra useState needed.
  const remaining = useMemo(() => todos.filter((t) => !t.done).length, [todos]);

  function toggle(id: number): void {
    setTodos((prev) => prev.map((t) => (t.id === id ? { ...t, done: !t.done } : t)));
  }

  function add(): void {
    const text = draft.trim();
    if (!text) return;
    setTodos((prev) => [...prev, { id: Date.now(), text, done: false }]);

    <section>
      <h2>{title}</h2>
      <p>{remaining} left</p>
      <div>
        <input
          value={draft}
          onChange={(e) => setDraft(e.target.value)}
          placeholder="New todo"
        />
        <button type="button" onClick={add}>
          Add
        </button>
      </div>
      <ul>
        {todos.map((todo) => (
          <TodoItem key={todo.id} todo={todo} onToggle={toggle} />
        ))}
      </ul>
    </section>
  );
}

  value: string;
  debounced: string;
  setValue: (next: string) => void;
}

// Reusable input debounce logic.
export function useDebounce({ delayMs, initial }: DebounceOptions): UseDebounceResult {
  const [value, setValue] = useState<string>(initial);
  const [debounced, setDebounced] = useState<string>(initial);

  useEffect(() => {
    const timer = window.setTimeout(() => setDebounced(value), delayMs);
    return () => window.clearTimeout(timer);
  }, [value, delayMs]);

  return { value, debounced, setValue };
}

  const [scrollTop, setScrollTop] = useState<number>(0);

  const visible = useMemo(() => {
    const start = Math.max(0, Math.floor(scrollTop / rowHeight) - 2);
    const count = Math.ceil(viewportHeight / rowHeight) + 4;
    return rows.slice(start, start + count).map((row, i) => ({
      row,
      index: start + i,
    }));

    <div
      onScroll={(e) => setScrollTop(e.currentTarget.scrollTop)}
      style={{ overflowY: 'auto', height: viewportHeight }}
    >
      <div style={{ position: 'relative', height: totalHeight }}>

          <RowView key={row.id} row={row} top={index * rowHeight} height={rowHeight} />
        ))}
      </div>

// Async helpers: AbortController, timeout, retry.
export function sleep(ms: number): Promise<void> {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

// Reject if the task takes longer than ms.
export async function withTimeout<T>(task: Promise<T>, ms: number): Promise<T> {
  let timer: ReturnType<typeof setTimeout> | undefined;
  const limit = new Promise<T>((_, reject) => {
    timer = setTimeout(() => reject(new Error("timed out")), ms);
  });
  try {
    return await Promise.race([task, limit]);
  } finally {
    if (timer) clearTimeout(timer);
  }
}

export interface RetryOpts {
  readonly tries: number;
  readonly delayMs: number;
}

// Retry a fallible async fn with a fixed delay.
export async function retry<T>(fn: () => Promise<T>, opts: RetryOpts): Promise<T> {
  let last: unknown;
  for (let i = 0; i < opts.tries; i++) {
    try {
      return await fn();
    } catch (e: unknown) {
      last = e;
      await sleep(opts.delayMs);
    }
  }
  if (last instanceof Error) throw last;
  throw new Error("retry failed");
}

// Fetch with an abort signal that fires after ms.
export async function fetchWithAbort(url: string, ms: number): Promise<Response> {
  const ctrl = new AbortController();
  const timer = setTimeout(() => ctrl.abort(), ms);
  try {
    return await fetch(url, { signal: ctrl.signal });
  } finally {
    clearTimeout(timer);
  }
}

// Run tasks with at most limit in flight.
export async function mapLimit<T, U>(
  items: readonly T[],
  limit: number,
  fn: (item: T) => Promise<U>
): Promise<U[]> {
  const out: U[] = new Array(items.length);
  let next = 0;
  async function worker(): Promise<void> {
    while (next < items.length) {
      const i = next;
      next += 1;
      const item = items[i];
      if (item !== undefined) out[i] = await fn(item);
    }
  }
  const workers: Promise<void>[] = [];
  const n = Math.min(limit, items.length);
  for (let k = 0; k < n; k++) workers.push(worker());
  await Promise.all(workers);
  return out;
}

function sameValue(a: unknown, b: unknown): boolean {
  return a === b || (Number.isNaN(a) && Number.isNaN(b));
}

// Deep clone plain data via JSON round trip.
export function deepClone<T>(value: T): T {
  return JSON.parse(JSON.stringify(value)) as T;
}

// Structural equality for plain data.
export function deepEqual(a: unknown, b: unknown): boolean {
  if (sameValue(a, b)) return true;
  if (typeof a !== typeof b) return false;
  if (typeof a !== "object" || a === null || b === null) return false;
  if (Array.isArray(a) || Array.isArray(b)) {
    if (!Array.isArray(a) || !Array.isArray(b)) return false;
    if (a.length !== b.length) return false;
    return a.every((x, i) => deepEqual(x, b[i]));
  }
  const ra = a as Record<string, unknown>;
  const rb = b as Record<string, unknown>;
  const ka = Object.keys(ra);
  const kb = Object.keys(rb);
  if (ka.length !== kb.length) return false;
  return ka.every((k) => deepEqual(ra[k], rb[k]));
}

// Unique values keeping first order.
export function unique<T>(items: readonly T[]): T[] {
  return [...new Set(items)];
}

// Split a list into fixed-size chunks.
export function chunk<T>(items: readonly T[], size: number): T[][] {
  const out: T[][] = [];
  for (let i = 0; i < items.length; i += size) {
    out.push(items.slice(i, i + size));
  }
  return out;
}

// Count occurrences by key.
export function countBy<T>(items: readonly T[], keyOf: (x: T) => string): Record<string, number> {
  const out: Record<string, number> = {};
  for (const x of items) {
    const k = keyOf(x);
    out[k] = (out[k] ?? 0) + 1;
  }
  return out;
}

// Config parsing with validation at the boundary.
export interface AppConfig {
  readonly host: string;
  readonly port: number;
  readonly debug: boolean;
  readonly tags: readonly string[];
}

export type ConfigError = { readonly field: string; readonly hint: string };
export type ConfigResult =
  | { readonly ok: true; readonly config: AppConfig }
  | { readonly ok: false; readonly errors: ConfigError[] };

function isRecord(v: unknown): v is Record<string, unknown> {
  return typeof v === "object" && v !== null && !Array.isArray(v);
}

const defaults = {
  host: "localhost",
  port: 3000,
  debug: false,
  tags: [],
} satisfies Record<keyof AppConfig, unknown>;

// Read one string field with a fallback.
function readString(src: Record<string, unknown>, key: string, fb: string): string {
  const v: unknown = src[key];
  return typeof v === "string" && v.length > 0 ? v : fb;
}

// Parse unknown input into a checked AppConfig.
export function parseConfig(input: unknown): ConfigResult {
  if (!isRecord(input)) {
    return { ok: false, errors: [{ field: "root", hint: "not an object" }] };
  }
  const errors: ConfigError[] = [];
  const host = readString(input, "host", defaults.host);

  let port = defaults.port;
  const rawPort: unknown = input["port"];
  if (typeof rawPort === "number" && Number.isInteger(rawPort)) {
    if (rawPort < 1 || rawPort > 65535) {
      errors.push({ field: "port", hint: "out of range" });
    } else {
      port = rawPort;
    }
  } else if (rawPort !== undefined) {
    errors.push({ field: "port", hint: "must be an integer" });
  }

  const rawDebug: unknown = input["debug"];
  const debug = typeof rawDebug === "boolean" ? rawDebug : defaults.debug;

  const rawTags: unknown = input["tags"];
  let tags: readonly string[] = defaults.tags;
  if (rawTags !== undefined) {
    if (Array.isArray(rawTags) && rawTags.every((x) => typeof x === "string")) {
      tags = rawTags as string[];
    } else {
      errors.push({ field: "tags", hint: "must be string[]" });
    }
  }

  if (errors.length > 0) return { ok: false, errors };
  return { ok: true, config: { host, port, debug, tags } };
}

export function defaultConfig(): AppConfig {
  return { ...defaults, tags: [...defaults.tags] };
}

// Typed event emitter keyed by event name.
export type Handler<T> = (payload: T) => void;

export interface AppEvents {
  readonly message: { readonly text: string };
  readonly count: { readonly n: number };
  readonly closed: { readonly reason: string };
}

export class Emitter<TEvents extends Record<string, unknown>> {
  private handlers = new Map<keyof TEvents, Set<Handler<never>>>();

  on<K extends keyof TEvents>(event: K, fn: Handler<TEvents[K]>): () => void {
    let set = this.handlers.get(event);
    if (!set) {
      set = new Set();
      this.handlers.set(event, set);
    }
    set.add(fn as Handler<never>);
    return () => this.off(event, fn);
  }

  off<K extends keyof TEvents>(event: K, fn: Handler<TEvents[K]>): void {
    this.handlers.get(event)?.delete(fn as Handler<never>);
  }

  once<K extends keyof TEvents>(event: K, fn: Handler<TEvents[K]>): void {
    const wrap: Handler<TEvents[K]> = (p) => {
      this.off(event, wrap);
      fn(p);
    };
    this.on(event, wrap);
  }

  emit<K extends keyof TEvents>(event: K, payload: TEvents[K]): void {
    const set = this.handlers.get(event);
    if (!set) return;
    for (const fn of [...set]) (fn as Handler<TEvents[K]>)(payload);
  }

  listenerCount<K extends keyof TEvents>(event: K): number {
    return this.handlers.get(event)?.size ?? 0;
  }
}

export type AppEmitter = Emitter<AppEvents>;

export function createAppEmitter(): AppEmitter {
  return new Emitter<AppEvents>();
}

// Emit a chat message event.
export function sendMessage(bus: AppEmitter, text: string): void {
  bus.emit("message", { text });
}

    this.name = "NetworkError";
  }
}

export type FetchOk<T> = { readonly ok: true; readonly data: T };
export type FetchFail = { readonly ok: false; readonly error: HttpError | NetworkError };
export type FetchResult<T> = FetchOk<T> | FetchFail;

export interface RequestOpts {
  readonly method?: HttpMethod;
  readonly headers?: Record<string, string>;
  readonly body?: string;
  readonly signal?: AbortSignal;
}

function isJsonObject(v: unknown): v is Record<string, unknown> {
  return typeof v === "object" && v !== null && !Array.isArray(v);
}

// Fetch JSON and validate with a guard fn.
export async function fetchJson<T>(
  url: string,
  guard: (v: unknown) => v is T,
  opts: RequestOpts = {}
): Promise<FetchResult<T>> {
  let res: Response;
  try {
    res = await fetch(url, {
      method: opts.method ?? "GET",
      headers: opts.headers,
      body: opts.body,
      signal: opts.signal,
    });
  } catch (e: unknown) {
    if (e instanceof Error) return { ok: false, error: new NetworkError(e.message) };
    return { ok: false, error: new NetworkError("fetch failed") };
  }
  if (!res.ok) {
    const text = await res.text();
    return { ok: false, error: new HttpError(res.status, text) };
  }
  const data: unknown = await res.json();
  void isJsonObject;
  if (guard(data)) return { ok: true, data };
  return { ok: false, error: new HttpError(res.status, "bad payload") };
}

// POST a JSON body and read a typed reply.
export async function postJson<T>(
  url: string,
  payload: Record<string, unknown>,
  guard: (v: unknown) => v is T
): Promise<FetchResult<T>> {
  return fetchJson(url, guard, {
    method: "POST",
    headers: { "content-type": "application/json" },
    body: JSON.stringify(payload),
  });
}

// Group items by a derived key.
export function groupBy<T, K extends string | number>(
  items: readonly T[],
  keyOf: (item: T) => K
): Record<K, T[]> {
  const out = {} as Record<K, T[]>;
  for (const item of items) {
    const key = keyOf(item);
    const slot = out[key];
    if (slot) slot.push(item);
    else out[key] = [item];
  }
  return out;
}

// Pick the item with the smallest score.
export function minBy<T>(items: readonly T[], score: (item: T) => number): T | undefined {
  let best: T | undefined;
  let bestScore = Infinity;
  for (const item of items) {
    const s = score(item);
    if (s < bestScore) {
      bestScore = s;
      best = item;
    }
  }
  return best;
}

export function sortBy<T>(items: readonly T[], key: (item: T) => number): T[] {
  return [...items].sort((a, b) => key(a) - key(b));
}

// Result type for explicit error handling.
export type Ok<T> = { readonly ok: true; readonly value: T };
export type Err<E> = { readonly ok: false; readonly error: E };
export type Result<T, E> = Ok<T> | Err<E>;

export function ok<T>(value: T): Ok<T> {
  return { ok: true, value };
}

export function err<E>(error: E): Err<E> {
  return { ok: false, error };
}

// Map the success value, keep the error as is.
export function map<T, U, E>(r: Result<T, E>, fn: (v: T) => U): Result<U, E> {
  if (!r.ok) return r;
  return ok(fn(r.value));
}

// Chain fallible steps without try/catch.
export function flatMap<T, U, E>(
  r: Result<T, E>,
  fn: (v: T) => Result<U, E>
): Result<U, E> {
  if (!r.ok) return r;
  return fn(r.value);
}

// Fallback value for the error case.
export function unwrapOr<T, E>(r: Result<T, E>, fallback: T): T {
  return r.ok ? r.value : fallback;
}

// Run a throwing fn and capture the error.
export function fromThrowable<T>(fn: () => T, label: string): Result<T, string> {
  try {
    return ok(fn());
  } catch (e: unknown) {
    if (e instanceof Error) return err(label + ": " + e.message);
    return err(label + ": unknown failure");
  }
}

// Combine many results, stop at first error.
export function combine<T, E>(items: Result<T, E>[]): Result<T[], E> {
  const out: T[] = [];
  for (const item of items) {
    if (!item.ok) return item;
    out.push(item.value);
  }
  return ok(out);
}

// Narrow a Result to its Ok branch.
export function isOk<T, E>(r: Result<T, E>): r is Ok<T> {
  return r.ok;
}

// Debounce rapid calls into one trailing call.
export function debounce<A extends unknown[]>(
  fn: (...args: A) => void,
  ms: number
): (...args: A) => void {
  let timer: ReturnType<typeof setTimeout> | undefined;
  return (...args: A) => {
    if (timer) clearTimeout(timer);
    timer = setTimeout(() => fn(...args), ms);
  };
}

// Throttle to at most one call per window.
export function throttle<A extends unknown[]>(
  fn: (...args: A) => void,
  ms: number
): (...args: A) => void {
  let last = 0;
  return (...args: A) => {
    const now = Date.now();
    if (now - last >= ms) {
      last = now;
      fn(...args);
    }
  };
}
